# Project Config
############################################################
file(GLOB_RECURSE SOURCE_FILES src/*.cpp)
file(GLOB_RECURSE HEADER_FILES src/*.hpp)
set(IMGUI_SOURCES
  depends/imgui/imgui_draw.cpp
  depends/imgui/imgui_tables.cpp
  depends/imgui/imgui_widgets.cpp
  depends/imgui/imgui.cpp
  depends/imgui/backends/imgui_impl_sdl3.cpp
  depends/imgui/backends/imgui_impl_sdlgpu3.cpp
  depends/imgui/backends/imgui_impl_sdlrenderer3.cpp
  depends/imgui/misc/cpp/imgui_stdlib.cpp
)

add_executable(${PROJECT_NAME} ${SOURCE_FILES} ${IMGUI_SOURCES})
target_include_directories(${PROJECT_NAME}
  PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/depends/imgui/
  PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/depends/imgui/backends/
)
target_compile_definitions(${PROJECT_NAME}
  PRIVATE IMGUI_DISABLE_OBSOLETE_FUNCTIONS
  PRIVATE IMGUI_DISABLE_DEMO_WINDOWS
  PRIVATE IMGUI_DISABLE_DEBUG_TOOLS
  )

source_group(TREE ${CMAKE_CURRENT_SOURCE_DIR}/src PREFIX "Source Files" FILES ${SOURCE_FILES})
source_group(TREE ${CMAKE_CURRENT_SOURCE_DIR}/src PREFIX "Header Files" FILES ${HEADER_FILES})

target_link_libraries(${PROJECT_NAME}
  PRIVATE SDL3-static
  PRIVATE SDL3_image-static
  PRIVATE SDL3_mixer-static
)
